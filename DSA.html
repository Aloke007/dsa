Practical 1: Linear & Binary Search

Linear Search scans every element sequentially (O(n)) and works on any array[cite: 22]. Binary Search repeatedly halves the search space (O(log n)) by comparing the target with the midpoint, requiring the array to be sorted[cite: 22].

// Algorithms[cite: 22]
// Linear: For i=0 to n-1: If arr[i] == key Return i. Return -1
// Binary: low=0, high=n-1. While low <= high: mid=(low+high)/2. If arr[mid]==key Return mid. Else adjust low/high.

// Shortest C Code[cite: 22, 23]
#include <stdio.h>
int linearSearch(int arr[], int n, int key) {
    for(int i=0; i<n; i++) if(arr[i] == key) return i;
    return -1;
}
int binarySearch(int arr[], int n, int key) {
    int low = 0, high = n-1;
    while(low <= high) {
        int mid = (low + high) / 2;
        if(arr[mid] == key) return mid;
        if(key < arr[mid]) high = mid - 1;
        else low = mid + 1;
    }
    return -1;
}
Practical 2: Bubble & Selection Sort

Bubble Sort repeatedly compares and swaps adjacent elements, bubbling the largest to the end (O(n^2), Stable)[cite: 24]. Selection Sort finds the minimum element from the unsorted portion and swaps it directly into its correct position (O(n^2), Unstable)[cite: 24].

// Algorithms[cite: 24]
// Bubble: For i=0 to n-2: For j=0 to n-i-2: If arr[j]>arr[j+1] Swap
// Selection: For i=0 to n-2: min_idx=i, For j=i+1 to n-1: If arr[j]<arr[min_idx] min_idx=j. Swap.

// Shortest C Code[cite: 24, 25]
#include <stdio.h>
void swap(int *a, int *b) { int t=*a; *a=*b; *b=t; }
void bubbleSort(int arr[], int n) {
    for(int i=0; i<n-1; i++)
        for(int j=0; j<n-i-1; j++)
            if(arr[j] > arr[j+1]) swap(&arr[j], &arr[j+1]);
}
void selectionSort(int arr[], int n) {
    for(int i=0; i<n-1; i++) {
        int min_idx = i;
        for(int j=i+1; j<n; j++)
            if(arr[j] < arr[min_idx]) min_idx = j;
        swap(&arr[i], &arr[min_idx]);
    }
}
Practical 3: Hash Table

Uses a hash function h1(k) = k mod m to map keys directly to array indices (O(1))[cite: 26]. Linear Probing checks the next sequential slot, while Double Hashing uses a second function as a jump step to distribute elements evenly[cite: 26].

// Shortest C Code[cite: 26, 27]
#define SIZE 10
int table[SIZE];
void init() { for(int i=0; i<SIZE; i++) table[i] = -1; }
void insertLinear(int key) {
    for(int i=0; i<SIZE; i++) {
        int probe = (key % SIZE + i) % SIZE;
        if(table[probe] == -1) { table[probe] = key; return; }
    }
}
void insertDoubleHash(int key) {
    int step = 7 - (key % 7);
    for(int i=0; i<SIZE; i++) {
        int probe = (key % SIZE + i * step) % SIZE;
        if(table[probe] == -1) { table[probe] = key; return; }
    }
}
Practical 4: Singly Linked List

Manage Pinnacle Club members. Insert President at beginning, Secretary at end, and delete members[cite: 32].

// C Code[cite: 32, 33]
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
typedef struct Node { int prn; char name[50]; struct Node* next; } Node;
Node* createNode(int prn, const char* name) {
    Node* newNode = (Node*)malloc(sizeof(Node));
    newNode->prn = prn; strcpy(newNode->name, name); newNode->next = NULL; return newNode;
}
Node* addPresident(Node* head, int prn, const char* name) {
    Node* newNode = createNode(prn, name); newNode->next = head; return newNode;
}
Node* addSecretary(Node* head, int prn, const char* name) {
    Node* newNode = createNode(prn, name);
    if (head == NULL) return newNode;
    Node* temp = head; while (temp->next != NULL) temp = temp->next;
    temp->next = newNode; return head;
}
Node* deleteMember(Node* head, int prn) {
    if (head == NULL) return NULL;
    Node* temp = head; Node* prev = NULL;
    while (temp != NULL && temp->prn != prn) { prev = temp; temp = temp->next; }
    if (temp == NULL) return head;
    if (prev != NULL) prev->next = temp->next; else head = temp->next;
    free(temp); return head;
}
Practical 5: Circular Doubly Linked List

Cinemax Theater Ticket Booking. Links seats circularly with previous and next pointers[cite: 34].

// C Code[cite: 34, 35]
#include <stdio.h>
#include <stdlib.h>
#define ROWS 10
#define SEATS 7
typedef struct Seat { int seatNo; int isBooked; struct Seat *prev, *next; } Seat;
Seat* headRow[ROWS];
void initTheater() {
    for (int r=0; r < ROWS; r++) {
        Seat *head = NULL, *tail = NULL;
        for (int i=1; i <= SEATS; i++) {
            Seat* newSeat = (Seat*)malloc(sizeof(Seat));
            newSeat->seatNo = i; newSeat->isBooked = 0;
            if (head == NULL) { head = tail = newSeat; head->next = head->prev = head; }
            else { tail->next = newSeat; newSeat->prev = tail; newSeat->next = head; head->prev = newSeat; tail = newSeat; }
        }
        headRow[r] = head;
    }
}
void bookSeat(int row, int seatNum) {
    Seat* temp = headRow[row - 1];
    do {
        if (temp->seatNo == seatNum) {
            if(temp->isBooked) printf("Booked!\n");
            else { temp->isBooked = 1; printf("Success.\n"); }
            return;
        }
        temp = temp->next;
    } while (temp != headRow[row - 1]);
}
Practical 6: Stack Applications

A Stack strictly follows Last-In-First-Out (LIFO)[cite: 28]. Conversion uses a character stack based on operator precedence, evaluation pushes numbers and applies operations to popped pairs[cite: 28].

// Shortest C Code[cite: 28, 29]
#include <ctype.h>
char cStack[100]; int cTop = -1;
void pushC(char c) { cStack[++cTop] = c; }
char popC() { return cStack[cTop--]; }
int pre(char op) { return (op=='^')?3 : (op=='*' || op=='/')?2 : (op=='+' || op=='-')?1 : 0; }

void infixToPostfix(char in[], char out[]) {
    int k=0; cTop=-1;
    for(int i=0; in[i]; i++) {
        if(isalnum(in[i])) out[k++] = in[i];
        else if(in[i]=='(') pushC('(');
        else if(in[i]==')') { while(cStack[cTop] != '(') out[k++] = popC(); popC(); }
        else {
            while(cTop != -1 && cStack[cTop] != '(' && pre(in[i]) <= pre(cStack[cTop]))
                out[k++] = popC();
            pushC(in[i]);
        }
    }
    while(cTop != -1) out[k++] = popC();
    out[k] = '\0';
}
Practical 7: Circular Queue

A Queue strictly follows First-In-First-Out (FIFO)[cite: 30]. Circular Queues wrap pointers using modulo arithmetic to reuse freed space[cite: 30].

// Shortest C Code[cite: 30, 31]
#define MAX 5
int Q[MAX], front = -1, rear = -1;
void enqueue(int val) {
    if((rear + 1) % MAX == front) return; // Full
    if(front == -1) front = 0;
    rear = (rear + 1) % MAX;
    Q[rear] = val;
}
int dequeue() {
    if(front == -1) return -1; // Empty
    int val = Q[front];
    if(front == rear) { front = -1; rear = -1; }
    else front = (front + 1) % MAX;
    return val;
}
Practical 8: Binary Search Tree (BST)

Recursive insertion handles left subtrees for smaller values and right subtrees for larger ones[cite: 35]. Mirroring recursively swaps left and right pointers[cite: 35].

// C Code[cite: 36, 37]
#include <stdio.h>
#include <stdlib.h>
typedef struct BSTNode { int data; struct BSTNode *left, *right; } BSTNode;
BSTNode* createNode(int value) {
    BSTNode* newNode = (BSTNode*)malloc(sizeof(BSTNode));
    newNode->data = value; newNode->left = newNode->right = NULL; return newNode;
}
BSTNode* insert(BSTNode* root, int value) {
    if (root == NULL) return createNode(value);
    if (value < root->data) root->left = insert(root->left, value);
    else if (value > root->data) root->right = insert(root->right, value);
    return root;
}
void mirrorTree(BSTNode* root) {
    if (root == NULL) return;
    mirrorTree(root->left); mirrorTree(root->right);
    BSTNode* temp = root->left; root->left = root->right; root->right = temp;
}
Practical 9: Graph (DFS)

DFS traversal recursively visits unvisited neighbors to check graph connectivity[cite: 37]. If any node remains unvisited after a full pass from the root, it is disconnected[cite: 37].

// C Code[cite: 37, 38]
#include <stdio.h>
#include <stdlib.h>
#define MAX_CITIES 20
typedef struct FlightNode { int destIndex; int cost; struct FlightNode* next; } FlightNode;
typedef struct { int numCities; FlightNode* adjList[MAX_CITIES]; } FlightGraph;
int visited[MAX_CITIES];

void DFS(FlightGraph* g, int u) {
    visited[u] = 1;
    FlightNode* temp = g->adjList[u];
    while (temp != NULL) {
        if (!visited[temp->destIndex]) DFS(g, temp->destIndex);
        temp = temp->next;
    }
}
int isConnected(FlightGraph* g) {
    for (int i=0; i < g->numCities; i++) visited[i] = 0;
    DFS(g, 0); 
    for (int i=0; i < g->numCities; i++) if (!visited[i]) return 0;
    return 1; 
}
